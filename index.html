MovementController.client.luau 
--!strict
-- [LocalScript] MovementController.client.luau
local UserInputService = game:GetService("UserInputService")
local player = game.Players.LocalPlayer
local character = player.Character or player.CharacterAdded:Wait()
local humanoid = character:WaitForChild("Humanoid") :: Humanoid
local rootPart = character:WaitForChild("HumanoidRootPart") :: BasePart

local canDoubleJump, hasDoubleJumped = false, false
local DASH_FORCE, DASH_COOLDOWN, lastDash = 80, 1.5, 0

humanoid.StateChanged:Connect(function(_, newState)
    if newState == Enum.HumanoidStateType.Landed then
        canDoubleJump = false
        hasDoubleJumped = false
    elseif newState == Enum.HumanoidStateType.Freefall then
        task.wait(0.1)
        canDoubleJump = true
    end
end)

UserInputService.InputBegan:Connect(function(input, processed)
    if processed then return end
    if input.KeyCode == Enum.KeyCode.Space and canDoubleJump and not hasDoubleJumped then
        hasDoubleJumped = true
        rootPart.AssemblyLinearVelocity = Vector3.new(rootPart.AssemblyLinearVelocity.X, 50, rootPart.AssemblyLinearVelocity.Z)
    elseif input.KeyCode == Enum.KeyCode.Q then
        if os.clock() - lastDash >= DASH_COOLDOWN then
            lastDash = os.clock()
            local dir = humanoid.MoveDirection.Magnitude > 0 and humanoid.MoveDirection or rootPart.CFrame.LookVector
            rootPart.AssemblyLinearVelocity = dir * DASH_FORCE + Vector3.new(0, 10, 0)
        end
    end
end)
--!strict
-- [ServerScript] BossAlertTrigger.server.luau
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Lighting = game:GetService("Lighting")
local TweenService = game:GetService("TweenService")

local trigger = script.Parent :: BasePart
local alertEvent = ReplicatedStorage:WaitForChild("BossAlertEvent") :: RemoteEvent
local triggered = false

trigger.Touched:Connect(function(hit)
    if triggered then return end
    local player = game.Players:GetPlayerFromCharacter(hit.Parent)
    if player then
        triggered = true
        alertEvent:FireAllClients({ Message = "⚠️ BOSS IS COMING! ⚠️", Duration = 4.5 })
        
        local atmosphere = Lighting:FindFirstChildOfClass("Atmosphere")
        if atmosphere then
            TweenService:Create(atmosphere, TweenInfo.new(2), {
                Color = Color3.fromRGB(180, 20, 20),
                Density = 0.45
            }):Play()
        end
    end
end)
--!strict
-- [ServerScript] EnvironmentalHazardLoop.server.luau
local hazardPart = script.Parent :: BasePart
local DAMAGE_TICK, TICK_RATE = 15, 0.5
local lastHitTable: { [Humanoid]: number } = {}

hazardPart.Touched:Connect(function(hit)
    local humanoid = hit.Parent:FindFirstChildOfClass("Humanoid")
    if humanoid and humanoid.Health > 0 then
        local now = os.clock()
        if not lastHitTable[humanoid] or (now - lastHitTable[humanoid] >= TICK_RATE) then
            lastHitTable[humanoid] = now
            humanoid:TakeDamage(DAMAGE_TICK)
        end
    end
end)
